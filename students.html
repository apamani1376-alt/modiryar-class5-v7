// بانک اطلاعات دانش آموزان مدیریار کلاس پنجم const students = []; // ساخت 32 دانش آموز در 8 گروه 4 نفره for(let i = 1; i <= 32; i++){ let group = Math.ceil(i / 4); students.push({ id: i, name: "دانش‌آموز " + i, group: group, // وضعیت آموزشی level: "", reading: "", writing: "", // ویژگی‌های فردی concentration: "", responsibility: "", roleTaking: "", // شناخت فردی interests: "", strengths: "", weaknesses: "", // وضعیت نیازمند توجه attentionNeeded: false, // یادداشت‌های معلم notes: "", // پایش یادگیری assessments: [], // امتیازات لیگ points: 0 }); } // پیدا کردن دانش آموز function getStudent(id){ return students.find( student => student.id === id ); } // اضافه کردن دانش آموز function addStudent(name){ let id = students.length + 1; students.push({ id:id, name:name, group:Math.ceil(id / 4), level:"", reading:"", writing:"", concentration:"", responsibility:"", roleTaking:"", interests:"", strengths:"", weaknesses:"", attentionNeeded:false, notes:"", assessments:[], points:0 }); }
